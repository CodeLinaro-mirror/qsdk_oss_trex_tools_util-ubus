net_pppoe.sh
parameter :
pre_del "delete old network setting"
interface   "network section name"
ifname  "network device name"
username    "pppoe username"
passwd  "pppoe password"
mtu 
disabled    "disable network"

example :
pre_del=1 interface=wan ifname=eth0 mtu=1500 username=pppoe passwd=12345678 disabled=0 ./net_pppoe.sh

interface   "network section name"
ifname  "network device name"
mtu

example :
pre_del=1 interface=wan ifname=eth0 mtu=1500 disabled=0 ./net_dhcp.sh

net_static.sh
parameter:
pre_del "delete old network setting"
interface   "network section name"
ifname  "network device name"
ipaddr  "network ip address"
netmask "network netmask"
mtu
gateway "network gateway"
disabled    "disable network"
dns1
dns2

example :
pre_del=1 interface=wan ifname=eth0 ipaddr=192.168.33.200 netmask=255.255.255.0 gateway=192.168.33.1 mtu=1500 dns1=8.8.8.8 dns2=0.0.0.0 ./net_static.sh

net_bridge.sh
parameter:
pre_del "delete old network setting"
interface   "network section name"
ifname  "network device name"
ipaddr  "network ip address"
netmask "network netmask"
gateway "network gateway"
disabled    "disable network"

example :
pre_del=1 interface=lan0 ifname='eth0.2.0 eth0.3.0 eth0.4.0 eth0.5.0' ipaddr=192.168.1.1 netmask=255.255.255.0 gateway=192.168.1.254 disabled=0 ./net_bridge.sh

util_uci [-f filename]
add     <config> <section>
set     <config>.<section>[.<option>]=<value>
get     <config>.<section>[.<option>]
del     <config>.<section>[.<option>]
changes <config>
commit  [<config>]
restart

del ucitest.wan

set ucitest.wan.netmask=255.255.255.2

set ucitest.wan.ipaddr=172.20.1.69

commit
restart
EOF
